<script>
document.addEventListener("DOMContentLoaded", function () {
    // La bienvenida REAL de esta página usa la clase .welcome.
    const welcome = document.querySelector(".welcome");
    if (!welcome) return;

    // Nunca se oculta automáticamente.
    welcome.style.animation = "none";
    welcome.style.animationDelay = "0s";
    welcome.style.opacity = "1";
    welcome.style.visibility = "visible";
    welcome.style.display = "flex";

    // Solo desaparece cuando el cliente pulsa "ENTRAR A LA TIENDA".
    const enterButton = welcome.querySelector(".enter-btn");
    if (enterButton) {
        enterButton.addEventListener("click", function (event) {
            event.preventDefault();

            welcome.style.transition = "opacity .6s ease, visibility .6s ease";
            welcome.style.opacity = "0";
            welcome.style.visibility = "hidden";

            setTimeout(function () {
                welcome.style.display = "none";
                document.querySelector("#inicio")?.scrollIntoView({behavior: "smooth"});
            }, 600);
        });
    }
});
</script>
